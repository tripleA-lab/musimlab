/************************************************* * MUSIMLAB ADMIN API * VERSION 4 *************************************************/ const SHEET_EVENTS = "EVENTS"; const SHEET_SEASONS = "Seasons"; const SHEET_PREPARATION = "Preparation"; const SHEET_LINK = "Link"; /************************************************* * GET *************************************************/ function doGet(e) { try { const action = e && e.parameter ? e.parameter.action : ""; if (action === "getSeasons") { return getSeasons(); } if (action === "getPreparation") { return getPreparation(e.parameter.eventId); } if (action === "getLinks") { return getLinks(e.parameter.eventId); } return getEvents(); } catch (error) { return jsonOutput({ success: false, message: error.toString() }); } } /************************************************* * POST *************************************************/ function doPost(e) { try { if (!e || !e.postData || !e.postData.contents) { return jsonOutput({ success: false, message: "POST data tidak diterima." }); } const request = JSON.parse(e.postData.contents); const action = request.action; const data = request.data || {}; if (!action) { return jsonOutput({ success: false, message: "Action tidak dihantar." }); } switch (action) { /* EVENTS */ case "addEvent": return addEvent(data); case "updateEvent": return updateEvent(data); case "deleteEvent": return deleteEvent(data); /* SEASONS */ case "addSeason": return addSeason(data); case "updateSeason": return updateSeason(data); case "deleteSeason": return deleteSeason(data); case "toggleSeason": return toggleSeason(data); /* PREPARATION */ case "addPreparation": return addPreparation(data); case "updatePreparation": return updatePreparation(data); case "deletePreparation": return deletePreparation(data); /* LINKS */ case "addLink": return addLink(data); case "updateLink": return updateLink(data); case "deleteLink": return deleteLink(data); default: return jsonOutput({ success: false, message: "Action tidak dikenali: " + action }); } } catch (error) { return jsonOutput({ success: false, message: error.toString() }); } } /************************************************* * EVENTS * * A Event ID * B Event Name * C Start Date * D End Date * E Category * F Location * G Short Description * H Description * I Season ID * J Image URL * K Official Link * L Status * M Created * N Updated *************************************************/ function getEvents() { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_EVENTS); const data = sheet.getDataRange().getValues(); if (data.length <= 1) { return jsonOutput([]); } const headers = data.shift(); const events = data.map(row => { const event = {}; headers.forEach( (header, index) => { event[header] = row[index]; } ); return event; }); return jsonOutput(events); } function addEvent(data) { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_EVENTS); const eventId = "EVT" + new Date().getTime(); const now = new Date(); sheet.appendRow([ eventId, data.eventName || "", data.startDate || "", data.endDate || "", data.category || "", data.location || "", data.shortDescription || "", data.description || "", data.seasonId || "", data.imageUrl || "", data.officialLink || "", data.status || "Draft", now, now ]); return jsonOutput({ success: true, message: "Event berjaya ditambah.", eventId: eventId }); } function updateEvent(data) { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_EVENTS); const rows = sheet.getDataRange().getValues(); for ( let i = 1; i < rows.length; i++ ) { if ( String(rows[i][0]) === String(data.eventId) ) { const rowNumber = i + 1; sheet .getRange( rowNumber, 2, 1, 11 ) .setValues([[ data.eventName || "", data.startDate || "", data.endDate || "", data.category || "", data.location || "", data.shortDescription || "", data.description || "", data.seasonId || "", data.imageUrl || "", data.officialLink || "", data.status || "Draft" ]]); sheet .getRange(rowNumber, 14) .setValue(new Date()); return jsonOutput({ success: true, message: "Event berjaya dikemaskini." }); } } return jsonOutput({ success: false, message: "Event ID tidak dijumpai." }); } function deleteEvent(data) { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_EVENTS); const rows = sheet.getDataRange().getValues(); for ( let i = 1; i < rows.length; i++ ) { if ( String(rows[i][0]) === String(data.eventId) ) { deleteRelatedPreparations( data.eventId ); deleteRelatedLinks( data.eventId ); sheet.deleteRow(i + 1); return jsonOutput({ success: true, message: "Event berjaya dipadam." }); } } return jsonOutput({ success: false, message: "Event ID tidak dijumpai." }); } /************************************************* * SEASONS * * A Season ID * B Season Name * C Description * D Status * E Created * F Updated *************************************************/ function getSeasons() { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_SEASONS); const data = sheet.getDataRange().getValues(); if (data.length <= 1) { return jsonOutput([]); } const headers = data.shift(); const seasons = data.map(row => { const season = {}; headers.forEach( (header, index) => { season[header] = row[index]; } ); return season; }); return jsonOutput(seasons); } function addSeason(data) { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_SEASONS); const seasonId = "SEA" + new Date().getTime(); const now = new Date(); sheet.appendRow([ seasonId, data.seasonName || "", data.description || "", data.status || "Hidden", now, now ]); return jsonOutput({ success: true, message: "Season berjaya ditambah.", seasonId: seasonId }); } function updateSeason(data) { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_SEASONS); const rows = sheet.getDataRange().getValues(); for ( let i = 1; i < rows.length; i++ ) { if ( String(rows[i][0]) === String(data.seasonId) ) { const rowNumber = i + 1; sheet .getRange( rowNumber, 2, 1, 3 ) .setValues([[ data.seasonName || "", data.description || "", data.status || "Hidden" ]]); sheet .getRange(rowNumber, 6) .setValue(new Date()); return jsonOutput({ success: true, message: "Season berjaya dikemaskini." }); } } return jsonOutput({ success: false, message: "Season ID tidak dijumpai." }); } /************************************************* * DELETE SEASON *************************************************/ function deleteSeason(data) { const seasonId = String(data.seasonId || "").trim(); if (!seasonId) { return jsonOutput({ success: false, message: "Season ID kosong." }); } const eventSheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_EVENTS); const eventRows = eventSheet .getDataRange() .getValues(); /* * Check whether the season * is actually being used. */ for ( let i = 1; i < eventRows.length; i++ ) { const eventSeasonId = String( eventRows[i][8] || "" ).trim(); if ( eventSeasonId && eventSeasonId === seasonId ) { return jsonOutput({ success: false, message: "Season masih digunakan oleh event. Tukarkan Season event tersebut dahulu." }); } } const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_SEASONS); const rows = sheet.getDataRange().getValues(); for ( let i = 1; i < rows.length; i++ ) { if ( String(rows[i][0]).trim() === seasonId ) { sheet.deleteRow(i + 1); return jsonOutput({ success: true, message: "Season berjaya dipadam." }); } } return jsonOutput({ success: false, message: "Season ID tidak dijumpai." }); } /************************************************* * TOGGLE SEASON *************************************************/ function toggleSeason(data) { const seasonId = String(data.seasonId || "").trim(); if (!seasonId) { return jsonOutput({ success: false, message: "Season ID kosong." }); } const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_SEASONS); const rows = sheet.getDataRange().getValues(); for ( let i = 1; i < rows.length; i++ ) { if ( String(rows[i][0]).trim() === seasonId ) { const rowNumber = i + 1; sheet .getRange(rowNumber, 4) .setValue( data.status || "Hidden" ); sheet .getRange(rowNumber, 6) .setValue(new Date()); return jsonOutput({ success: true, message: "Status season berjaya dikemaskini." }); } } return jsonOutput({ success: false, message: "Season ID tidak dijumpai." }); } /************************************************* * PREPARATION * * A Preparation ID * B Event ID * C Item * D Reason *************************************************/ function getPreparation(eventId) { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_PREPARATION); const data = sheet.getDataRange().getValues(); if (data.length <= 1) { return jsonOutput([]); } const headers = data.shift(); const results = data .filter(row => String(row[1]) === String(eventId) ) .map(row => { const item = {}; headers.forEach( (header, index) => { item[header] = row[index]; } ); return item; }); return jsonOutput(results); } function addPreparation(data) { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_PREPARATION); const preparationId = "PREP" + new Date().getTime(); if (!data.eventId) { return jsonOutput({ success: false, message: "Event ID diperlukan." }); } if (!data.item) { return jsonOutput({ success: false, message: "Item diperlukan." }); } sheet.appendRow([ preparationId, data.eventId, data.item, data.reason || "" ]); return jsonOutput({ success: true, message: "Preparation berjaya ditambah.", preparationId: preparationId }); } function updatePreparation(data) { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_PREPARATION); const rows = sheet.getDataRange().getValues(); for ( let i = 1; i < rows.length; i++ ) { if ( String(rows[i][0]) === String(data.preparationId) ) { const rowNumber = i + 1; sheet .getRange( rowNumber, 3, 1, 2 ) .setValues([[ data.item || "", data.reason || "" ]]); return jsonOutput({ success: true, message: "Preparation berjaya dikemaskini." }); } } return jsonOutput({ success: false, message: "Preparation ID tidak dijumpai." }); } function deletePreparation(data) { const id = String( data.preparationId || "" ).trim(); if (!id) { return jsonOutput({ success: false, message: "Preparation ID kosong." }); } const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_PREPARATION); const rows = sheet.getDataRange().getValues(); for ( let i = 1; i < rows.length; i++ ) { if ( String(rows[i][0]).trim() === id ) { sheet.deleteRow(i + 1); return jsonOutput({ success: true, message: "Preparation berjaya dipadam." }); } } return jsonOutput({ success: false, message: "Preparation ID tidak dijumpai." }); } /************************************************* * LINKS * * A Link ID * B Event ID * C Link Type * D Title * E URL *************************************************/ function getLinks(eventId) { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_LINK); const data = sheet.getDataRange().getValues(); if (data.length <= 1) { return jsonOutput([]); } const headers = data.shift(); const results = data .filter(row => String(row[1]) === String(eventId) ) .map(row => { const link = {}; headers.forEach( (header, index) => { link[header] = row[index]; } ); return link; }); return jsonOutput(results); } function addLink(data) { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_LINK); const linkId = "LNK" + new Date().getTime(); if (!data.eventId) { return jsonOutput({ success: false, message: "Event ID diperlukan." }); } if (!data.title) { return jsonOutput({ success: false, message: "Title diperlukan." }); } if (!data.url) { return jsonOutput({ success: false, message: "URL diperlukan." }); } sheet.appendRow([ linkId, data.eventId, data.linkType || "", data.title, data.url ]); return jsonOutput({ success: true, message: "Link berjaya ditambah.", linkId: linkId }); } function updateLink(data) { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_LINK); const rows = sheet.getDataRange().getValues(); for ( let i = 1; i < rows.length; i++ ) { if ( String(rows[i][0]) === String(data.linkId) ) { const rowNumber = i + 1; sheet .getRange( rowNumber, 3, 1, 3 ) .setValues([[ data.linkType || "", data.title || "", data.url || "" ]]); return jsonOutput({ success: true, message: "Link berjaya dikemaskini." }); } } return jsonOutput({ success: false, message: "Link ID tidak dijumpai." }); } function deleteLink(data) { const id = String( data.linkId || "" ).trim(); if (!id) { return jsonOutput({ success: false, message: "Link ID kosong." }); } const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_LINK); const rows = sheet.getDataRange().getValues(); for ( let i = 1; i < rows.length; i++ ) { if ( String(rows[i][0]).trim() === id ) { sheet.deleteRow(i + 1); return jsonOutput({ success: true, message: "Link berjaya dipadam." }); } } return jsonOutput({ success: false, message: "Link ID tidak dijumpai." }); } /************************************************* * DELETE RELATED RECORDS *************************************************/ function deleteRelatedPreparations(eventId) { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_PREPARATION); const data = sheet.getDataRange().getValues(); for ( let i = data.length - 1; i >= 1; i-- ) { if ( String(data[i][1]) === String(eventId) ) { sheet.deleteRow(i + 1); } } } function deleteRelatedLinks(eventId) { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_LINK); const data = sheet.getDataRange().getValues(); for ( let i = data.length - 1; i >= 1; i-- ) { if ( String(data[i][1]) === String(eventId) ) { sheet.deleteRow(i + 1); } } } /************************************************* * JSON *************************************************/ function jsonOutput(data) { return ContentService .createTextOutput( JSON.stringify(data) ) .setMimeType( ContentService.MimeType.JSON ); }
